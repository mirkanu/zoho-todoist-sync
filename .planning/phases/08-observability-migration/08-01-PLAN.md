    Create `app/health/router.py` with this concrete structure:

    ```python
    from __future__ import annotations
    from datetime import datetime, timezone, timedelta
    from fastapi import APIRouter, Request
    from fastapi.responses import JSONResponse
    from sqlalchemy import select, func
    from arq.constants import default_queue_name, in_progress_key_prefix

    from app.core.logging import get_logger
    from app.db.models import SyncEvent, SyncState, KVStore
    from app.worker.reconciler import KV_RECONCILER_LAST_RUN

    log = get_logger(__name__)
    router = APIRouter()

    STALE_RECONCILER_MINUTES = 30
    DEGRADED_ERROR_THRESHOLD = 10

    def _compute_status(errors_24h: int, reconciler_last_run: str | None, queue_failed: int) -> str:
        if queue_failed > 0:
            return "error"
        if reconciler_last_run is None:
            return "error"
        try:
            last_run = datetime.fromisoformat(reconciler_last_run)
        except ValueError:
            return "error"
        if (datetime.now(timezone.utc) - last_run) > timedelta(minutes=STALE_RECONCILER_MINUTES):
            return "error"
        if errors_24h > DEGRADED_ERROR_THRESHOLD:
            return "degraded"
        return "ok"

    @router.get("/health")
    async def health(request: Request) -> JSONResponse:
        redis = request.app.state.redis
        session_factory = request.app.state.session_factory

        # Redis metrics — O(1) zcard + bounded scan_iter
        queue_depth = await redis.zcard(default_queue_name)
        in_progress = 0
        async for _ in redis.scan_iter(match=in_progress_key_prefix + "*"):
            in_progress += 1

        cutoff_24h = datetime.now(timezone.utc) - timedelta(hours=24)
        async with session_factory() as session:
            errors_24h = (await session.execute(
                select(func.count()).select_from(SyncEvent).where(
                    SyncEvent.action == "error",
                    SyncEvent.created_at > cutoff_24h,
                )
            )).scalar_one()

            echoes_24h = (await session.execute(
                select(func.count()).select_from(SyncEvent).where(
                    SyncEvent.action == "echo_suppressed",
                    SyncEvent.created_at > cutoff_24h,
                )
            )).scalar_one()

            syncs_24h = (await session.execute(
                select(func.count()).select_from(SyncEvent).where(
                    SyncEvent.action == "sync",
                    SyncEvent.created_at > cutoff_24h,
                )
            )).scalar_one()

            active_tasks = (await session.execute(
                select(func.count()).select_from(SyncState)
            )).scalar_one()

            last_event = (await session.execute(
                select(SyncEvent)
                .where(SyncEvent.action == "sync")
                .order_by(SyncEvent.created_at.desc())
                .limit(1)
            )).scalar_one_or_none()

            reconciler_kv = (await session.execute(
                select(KVStore.value).where(KVStore.key == KV_RECONCILER_LAST_RUN)
            )).scalar_one_or_none()

        # OBS-1: queue.failed proxy = errors_24h (research Pitfall 1: never iterate arq:result:*)
        queue_failed = errors_24h
        status = _compute_status(errors_24h, reconciler_kv, queue_failed)
        http_code = 503 if status == "error" else 200

        last_sync = None
        if last_event is not None:
            last_sync = {
                "at": last_event.created_at.isoformat(),
                "source": last_event.source,
            }

        body = {
            "status": status,
            "last_sync": last_sync,
            "queue": {
                "depth": queue_depth,
                "in_progress": in_progress,
                "failed": queue_failed,
            },
            "errors_24h": errors_24h,
            "echoes_suppressed_24h": echoes_24h,
            "syncs_24h": syncs_24h,
            "active_tasks": active_tasks,
            "reconciler": {"last_run": reconciler_kv},
        }
        return JSONResponse(content=body, status_code=http_code)
    ```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| External HTTP → /health | Unauthenticated endpoint exposed by FastAPI web service; consistent with existing webhooks router (no app-level auth — relies on Railway network policy) |
| /health handler → Postgres | Read-only count queries via SQLAlchemy parameterised statements |
| /health handler → Redis | Read-only zcard + scan_iter on arq-namespaced keys |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-08-01 | Information Disclosure | GET /health response body | accept | Response contains only counts and timestamps — no secrets, no task content, no PII. Active_tasks count and error counts are operational metadata only. Consistent with existing unauthenticated webhook endpoints. [Assumption A2 in RESEARCH.md acknowledges Railway network policy is the boundary control.] |
| T-08-02 | Denial of Service | GET /health (slow query exhaustion) | mitigate | Endpoint uses only O(1) count queries with indexed `created_at` column (existing index from INFRA-2) and Redis ZCARD (O(1)); `scan_iter` over `arq:in-progress:*` is bounded by concurrent worker count (small). Pitfall 1 guard: NEVER iterate `arq:result:*`. Verified by `test_health_uses_zcard_not_keys`. |
| T-08-03 | Tampering | SQL injection via /health | mitigate | All queries use SQLAlchemy ORM parameterised constructs (`select(func.count()).where(SyncEvent.action == "error")`). No string interpolation into SQL. No user input on this route (GET with no query params). |
| T-08-04 | Spoofing | Forged /health responses | accept | Read-only endpoint; no state changes; spoofing harmful only at the network layer (TLS termination at Railway). Out of scope for application code. |
</threat_model>
